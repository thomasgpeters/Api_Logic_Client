### 1
query = fn InitQueryDictionary
fn AddTableToQuery( query, @"person", @"p" )
fn AddColumnToQuery( query, @"p.id", @"ID" )
fn AddColumnToQuery( query, @"p.givenName", @"Given Name" )
fn AddColumnToQuery( query, @"p.surName", @"Surname" )
fn AddColumnToQuery( query, @"p.balance", @"Balance" )
sql = fn GenerateSelectSQL( query )
MutableStringAppendString( sql, @" ORDER BY \"p\".\"surName\"" )  // by hand
